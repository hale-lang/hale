// dna/org/work.hl for the dogfood run (GH #946): a deterministic
// performer standing in for the model. It takes the software kind, makes
// its change in a scratch clone of the project, commits it, and hands the
// commit back as the outcome's `result_ref` with its own patch as a
// receipt: the owner makes that patch a Mutation and its Review (GH #1156).
import "vendor/dna" as dna;
import "vendor/dna/legs" as legs;

locus Candidate {
    params { pause_ms: Int = 0; }
    fn kind() -> String { return "deterministic"; }
    // run again under a new lease it makes the same change in a new clone:
    // the same end, whatever a dead run left
    fn effect_class() -> String { return "idempotent"; }
    fn takes(performer_kind: String) -> Bool { return performer_kind == "software"; }
    fn failed(why: String, shown: String) -> legs::Performance {
        return legs::Performance { disposition: "failed", narrative: why, prompt_digest: shown, renderer: legs::RENDERER_VERSION };
    }
    @unbounded
    fn perform(brief: legs::Brief, hands: legs::Hands) -> legs::Performance {
        let shown = legs::prompt_digest(brief.rendered);
        let top = dna::run_tool("git\nrev-parse\n--show-toplevel");
        if top.code != 0 { return self.failed("not in the project's repository: " + top.stderr, shown); }
        let repo = dna::trim_nl(top.stdout);
        let clone = hands.git.worktree(repo, "HEAD");
        if !clone.ok { return self.failed("no scratch clone: " + clone.error, shown); }
        // the pause is where a leg dies mid-task in the recovery case: the
        // clone is made, and the marker says where, so the fixture kills it here
        if self.pause_ms > 0 {
            std::io::fs::write_file(repo + "/.hale/dogfood.cloned", clone.output) or discard;
            let until = std::time::monotonic_ns() + self.pause_ms * 1000000;
            while std::time::monotonic_ns() < until { std::time::sleep(100ms); }
        }
        let name = std::str::replace(brief.attempt_id, "/", "-");
        let patch = "--- /dev/null\n+++ b/api/notes/" + name + ".md\n@@ -0,0 +1 @@\n+" + brief.attempt_id + "\n";
        let applied = hands.git.apply(clone.output, patch);
        if !applied.ok { return self.failed("the change did not apply: " + applied.error, shown); }
        let committed = hands.git.commit(clone.output, "api: a note for " + brief.attempt_id);
        if !committed.ok { return self.failed("the change did not commit: " + committed.error, shown); }
        let sha = committed.output;
        let exported = hands.git.patch(clone.output);
        if !exported.ok { return self.failed("the candidate's patch was not exported: " + exported.error, shown); }
        return legs::Performance {
            disposition: "done", result: "candidate " + sha, result_ref: "commit:" + sha,
            narrative: "the change is commit " + sha,
            receipts: "[{\"class\": \"internal\", \"body\": " + dna::record_json_quote(exported.output) + "}]",
            prompt_digest: shown, renderer: legs::RENDERER_VERSION
        };
    }
}

fn performers() -> legs::PerformerCatalog {
    return legs::PerformerCatalog { person: legs::Person { effect: "idempotent" }, deterministic: Candidate { pause_ms: @@PAUSE@@ }, model: legs::NoModel { } };
}
fn hands() -> legs::Hands { return legs::Hands { }; }
